# ── Helper functions ──────────────────────────────────────────────────────────
function T([string]$key) {
    $lang = $script:Language
    if ($text.ContainsKey($lang) -and $text[$lang].ContainsKey($key)) { return $text[$lang][$key] }
    if ($text['EN'].ContainsKey($key)) { return $text['EN'][$key] }
    return "[$key]"
}
function Get-YesKey { return (T 'YesKey') }

function Pause-Menu {
    try { Read-Host "`n$(T 'Pause')" | Out-Null } catch { }
}

function Confirm-Exit {
    $yesKey = Get-YesKey
    try {
        $answer = (Read-Host "$(T 'ExitAsk') [$yesKey/N]").Trim().ToUpperInvariant()
        return $answer -eq $yesKey
    }
    catch { return $false }
}

# ── Modern Inline Progress Bar (never duplicates or wraps) ───────────────────
function Show-ModernProgressBar {
    param(
        [int]    $Current,
        [int]    $Total,
        [string] $StatusText = '',
        [string] $BarColor   = 'Cyan',
        [int]    $BarWidth   = 20
    )
    if ($Total -le 0) { return }
    $pct = [math]::Min(100, [math]::Max(0, [int](($Current / $Total) * 100)))
    $filledCount = [int](($BarWidth * $pct) / 100)
    $emptyCount = [math]::Max(0, $BarWidth - $filledCount)

    $filled = [string]::new([char]0x2588, $filledCount)  # █
    $empty  = [string]::new([char]0x2591, $emptyCount)   # ░

    # Keep line well below 80 columns (safe width 64) so it NEVER wraps or duplicates
    $safeWidth = 64
    $header = "  [$filled$empty] $($pct.ToString().PadLeft(3))% ($Current/$Total)"
    $maxStatusLen = [math]::Max(0, $safeWidth - $header.Length - 1)
    $displayStatus = $StatusText
    if ($displayStatus.Length -gt $maxStatusLen) {
        if ($maxStatusLen -gt 4) {
            $displayStatus = '...' + $displayStatus.Substring($displayStatus.Length - ($maxStatusLen - 3))
        } else {
            $displayStatus = ''
        }
    }

    $line = "$header $displayStatus".PadRight($safeWidth)
    if ($line.Length -gt $safeWidth) { $line = $line.Substring(0, $safeWidth) }

    try {
        $pos = $Host.UI.RawUI.CursorPosition
        $pos.X = 0
        $Host.UI.RawUI.CursorPosition = $pos
    } catch { }

    Write-Host "`r$line" -NoNewline -ForegroundColor $BarColor

    if ($Current -ge $Total) {
        Write-Host ""
    }
}

# ── Real-Time Download Progress Bar (never duplicates or wraps) ───────────────
function Show-DownloadProgressBar {
    param(
        [int64]  $BytesReceived,
        [int64]  $TotalBytes,
        [double] $SpeedMBps = 0.0,
        [int]    $BarWidth  = 16
    )
    if ($TotalBytes -le 0) {
        $mbRec = [math]::Round($BytesReceived / 1MB, 1)
        $speedStr = if ($SpeedMBps -gt 0) { "$([math]::Round($SpeedMBps, 1)) MB/s" } else { "-- MB/s" }
        $line = "  Descargando: $mbRec MB ($speedStr)".PadRight(64)
        if ($line.Length -gt 64) { $line = $line.Substring(0, 64) }
        try {
            $pos = $Host.UI.RawUI.CursorPosition
            $pos.X = 0
            $Host.UI.RawUI.CursorPosition = $pos
        } catch { }
        Write-Host "`r$line" -NoNewline -ForegroundColor Cyan
        return
    }

    $pct = [math]::Min(100, [math]::Max(0, [int](($BytesReceived / $TotalBytes) * 100)))
    $filledCount = [int](($BarWidth * $pct) / 100)
    $emptyCount = [math]::Max(0, $BarWidth - $filledCount)

    $filled = [string]::new([char]0x2588, $filledCount)  # █
    $empty  = [string]::new([char]0x2591, $emptyCount)   # ░

    $recGB = [math]::Round($BytesReceived / 1GB, 2)
    $totGB = [math]::Round($TotalBytes / 1GB, 2)
    $speedStr = if ($SpeedMBps -gt 0) { "$([math]::Round($SpeedMBps, 1)) MB/s" } else { "-- MB/s" }

    # Format: "  [████░░░░]  45% (3.84/8.54 GB) 12.5 MB/s"
    # Keep strictly <= 64 chars to avoid any wrap/duplication on resize or move
    $line = "  [$filled$empty] $($pct.ToString().PadLeft(3))% ($recGB/$totGB GB) $speedStr".PadRight(64)
    if ($line.Length -gt 64) { $line = $line.Substring(0, 64) }

    try {
        $pos = $Host.UI.RawUI.CursorPosition
        $pos.X = 0
        $Host.UI.RawUI.CursorPosition = $pos
    } catch { }

    Write-Host "`r$line" -NoNewline -ForegroundColor Cyan
}

# ── High-Speed Direct Streaming Download with Live Progress ───────────────────
function Invoke-DownloadDirectStream {
    param(
        [string] $Url,
        [string] $Destination
    )
    $req = [System.Net.HttpWebRequest]::Create($Url)
    $req.Method = 'GET'
    $req.UserAgent = 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)'
    $req.Timeout = 60000
    $req.ReadWriteTimeout = 60000
    $req.AllowAutoRedirect = $true

    $resp = $req.GetResponse()
    $totalBytes = $resp.ContentLength
    $stream = $resp.GetResponseStream()
    $destDir = Split-Path -Parent $Destination
    if (-not (Test-Path $destDir)) { New-Item -ItemType Directory -Force -Path $destDir | Out-Null }
    $fileStream = [System.IO.File]::Create($Destination)

    $buffer = New-Object byte[] (512 * 1024)
    $bytesReceived = [int64]0
    $sw = [System.Diagnostics.Stopwatch]::StartNew()
    $lastUpdate = [int64]0
    $lastBytes = [int64]0
    $currentSpeed = 0.0

    try {
        while (($read = $stream.Read($buffer, 0, $buffer.Length)) -gt 0) {
            $fileStream.Write($buffer, 0, $read)
            $bytesReceived += $read

            $elapsedMs = $sw.ElapsedMilliseconds
            if ($elapsedMs - $lastUpdate -ge 250) {
                $timeDeltaSec = ($elapsedMs - $lastUpdate) / 1000.0
                $bytesDelta = $bytesReceived - $lastBytes
                if ($timeDeltaSec -gt 0) {
                    $currentSpeed = ($bytesDelta / 1MB) / $timeDeltaSec
                }
                $lastUpdate = $elapsedMs
                $lastBytes = $bytesReceived
                Show-DownloadProgressBar -BytesReceived $bytesReceived -TotalBytes $totalBytes -SpeedMBps $currentSpeed
            }
        }
        Show-DownloadProgressBar -BytesReceived $bytesReceived -TotalBytes $totalBytes -SpeedMBps $currentSpeed
        Write-Host ""
    }
    finally {
        if ($fileStream) { $fileStream.Flush(); $fileStream.Close(); $fileStream.Dispose() }
        if ($stream) { $stream.Close(); $stream.Dispose() }
        if ($resp) { $resp.Close(); $resp.Dispose() }
    }
}

# ── Download Manager: BITS with live progress + fallback to Direct Stream ─────
function Invoke-DownloadWithProgressBar {
    param(
        [string] $Url,
        [string] $Destination
    )
    $bitsSucceeded = $false
    if (Get-Command Start-BitsTransfer -ErrorAction SilentlyContinue) {
        try {
            Write-Host " -> $(T 'DownloadMethod')" -ForegroundColor DarkCyan
            $job = Start-BitsTransfer -Source $Url -Destination $Destination -DisplayName 'Zombies Declassified' -Asynchronous -ErrorAction Stop
            $sw = [System.Diagnostics.Stopwatch]::StartNew()
            $lastBytes = [int64]0
            $lastMs = [int64]0
            $speed = 0.0

            while ($job.JobState -in @('Connecting', 'Transferring', 'TransientError')) {
                $bytes = [int64]$job.BytesTransferred
                $total = [int64]$job.BytesTotal
                $nowMs = $sw.ElapsedMilliseconds
                if ($nowMs - $lastMs -ge 350) {
                    $dt = ($nowMs - $lastMs) / 1000.0
                    if ($dt -gt 0) {
                        $speed = (($bytes - $lastBytes) / 1MB) / $dt
                    }
                    $lastBytes = $bytes
                    $lastMs = $nowMs
                    Show-DownloadProgressBar -BytesReceived $bytes -TotalBytes $total -SpeedMBps $speed
                }
                Start-Sleep -Milliseconds 200
            }

            if ($job.JobState -eq 'Transferred') {
                Complete-BitsTransfer -BitsJob $job
                Show-DownloadProgressBar -BytesReceived $job.BytesTransferred -TotalBytes $job.BytesTotal -SpeedMBps 0
                Write-Host ""
                $bitsSucceeded = $true
            } else {
                try { Remove-BitsTransfer -BitsJob $job -ErrorAction SilentlyContinue } catch { }
            }
        }
        catch {
            $bitsSucceeded = $false
        }
    }

    if (-not $bitsSucceeded -or -not (Test-Path -LiteralPath $Destination)) {
        if (Test-Path -LiteralPath $Destination) {
            Remove-Item -LiteralPath $Destination -Force -ErrorAction SilentlyContinue
        }
        Write-Host " -> $(T 'FallbackStream')" -ForegroundColor DarkCyan
        Invoke-DownloadDirectStream -Url $Url -Destination $Destination
    }
}

function Get-TempRoot {
    foreach ($candidate in @($env:TEMP, $env:TMP, [System.IO.Path]::GetTempPath())) {
        if (-not [string]::IsNullOrWhiteSpace($candidate)) {
            try {
                $full = [System.IO.Path]::GetFullPath($candidate)
                if (Test-Path $full) { return $full }
            }
            catch { }
        }
    }
    return $script:Root
}

function Get-FreeDiskSpace([string]$path) {
    try {
        $drive = Split-Path -Qualifier $path
        $letter = $drive.TrimEnd(':')
        $wmi = Get-WmiObject -Class Win32_LogicalDisk -Filter "DeviceID='${letter}:'" -ErrorAction SilentlyContinue
        if ($null -ne $wmi) { return [int64]$wmi.FreeSpace }
    }
    catch { }
    return [int64]::MaxValue
}

function Select-Folder([string]$description) {
    $env:ZDC_FOLDER_DESCRIPTION = $description
    try {
        $selected = & powershell.exe -NoProfile -STA -ExecutionPolicy Bypass -Command {
            Add-Type -AssemblyName System.Windows.Forms
            [System.Windows.Forms.Application]::EnableVisualStyles()
            $dialog = New-Object System.Windows.Forms.FolderBrowserDialog
            $dialog.Description = $env:ZDC_FOLDER_DESCRIPTION
            $dialog.ShowNewFolderButton = $false
            if ($dialog.ShowDialog() -eq [System.Windows.Forms.DialogResult]::OK) { $dialog.SelectedPath }
        }
    }
    catch { $selected = $null }
    finally { Remove-Item Env:ZDC_FOLDER_DESCRIPTION -ErrorAction SilentlyContinue }
    if ($selected) {
        $last = ($selected | Select-Object -Last 1).ToString().Trim()
        if (-not [string]::IsNullOrWhiteSpace($last)) { return $last }
    }
    return $null
}

# ── Package Auto-Detection & Verification ─────────────────────────────────────
function Test-PackageFolder([string]$folderPath) {
    if ([string]::IsNullOrWhiteSpace($folderPath) -or -not (Test-Path -LiteralPath $folderPath)) {
        return $false
    }
    $hasStorage = Test-Path (Join-Path $folderPath 'storage\t6')
    $hasZone = Test-Path (Join-Path $folderPath 'steam\zone')
    $hasSound = Test-Path (Join-Path $folderPath 'steam\sound')
    return ($hasStorage -and $hasZone -and $hasSound)
}

function Find-PackageAuto {
    # 1. Check current script root / selected root
    if (Test-PackageFolder $script:Root) {
        return $script:Root
    }
    # 2. Check Desktop folder
    $desktopPath = [System.Environment]::GetFolderPath('Desktop')
    $desktopPkg = Join-Path $desktopPath 'ZombiesDeclassified_BETA1'
    if (Test-PackageFolder $desktopPkg) {
        return $desktopPkg
    }
    # 3. Check current working directory
    if ($PSScriptRoot -and (Test-PackageFolder $PSScriptRoot)) {
        return $PSScriptRoot
    }
    return $null
}

function Select-Package {
    $autoPath = Find-PackageAuto
    if ($autoPath) {
        $script:Root = $autoPath
        $script:StorageSource = Join-Path $autoPath 'storage\t6'
        $script:GameSource = Join-Path $autoPath 'steam'
        Write-Host "`n$(T 'PackageAutoFound') $autoPath" -ForegroundColor Green
        return $true
    }

    # 2. If not found automatically, display MediaFire link and prompt options
    Write-Host "`n========================================================" -ForegroundColor Yellow
    Write-Host " $(T 'PackageNotFound')" -ForegroundColor Yellow
    Write-Host "========================================================" -ForegroundColor Yellow
    Write-Host "`n$(T 'MediaFireLinkHeader')" -ForegroundColor Cyan
    Write-Host " -> $script:MediaFireUrl" -ForegroundColor Gray
    Write-Host "`n$(T 'PackageMissingPrompt')" -ForegroundColor Yellow

    $opt = $null
    try { $opt = (Read-Host "$(T 'Select') [1/2/3]").Trim() } catch { }

    switch ($opt) {
        '1' {
            Write-Host "`n$(T 'Package')" -ForegroundColor Cyan
            Write-Host "$(T 'PackageHelp')"
            $selected = Select-Folder (T 'Package')
            if ([string]::IsNullOrWhiteSpace($selected)) {
                Write-Host "`n$(T 'CancelFolderReturn')" -ForegroundColor Yellow
                Start-Sleep -Milliseconds 800
                return $false
            }
            if (-not (Test-PackageFolder $selected)) {
                Write-Host "`n$(T 'InvalidPackage')" -ForegroundColor Red
                Start-Sleep -Milliseconds 1200
                return $false
            }
            $script:Root = $selected
            $script:StorageSource = Join-Path $selected 'storage\t6'
            $script:GameSource = Join-Path $selected 'steam'
            Write-Host "`n$(T 'PackageFound'): $selected" -ForegroundColor Green
            return $true
        }
        '2' {
            Clear-Host
            $downloaded = Invoke-DownloadPackage
            if ($downloaded) {
                return $true
            }
            return $false
        }
        default {
            Write-Host "`n$(T 'ReturningToMenu')" -ForegroundColor Yellow
            Start-Sleep -Milliseconds 600
            return $false
        }
    }
}

    $tempRoot = Get-TempRoot
    $zipPath = Join-Path $tempRoot 'ZombiesDeclassified_BETA1.zip'
    $extractRoot = Join-Path $tempRoot 'ZombiesDeclassified_BETA1_package'

function Invoke-DownloadPackage {
    Write-Host "`n========================================================" -ForegroundColor Cyan
    Write-Host " $(T 'Download')" -ForegroundColor Cyan
    Write-Host "========================================================" -ForegroundColor Cyan

    Write-Host "`n$(T 'MediaFireLinkHeader')" -ForegroundColor Cyan
    Write-Host " -> $script:MediaFireUrl" -ForegroundColor Gray

    $tempRoot = Get-TempRoot
    $zipPath = Join-Path $tempRoot 'ZombiesDeclassified_BETA1.zip'
    $extractRoot = Join-Path $tempRoot 'ZombiesDeclassified_BETA1_package'

    # Check disk space
    $freeBytes = Get-FreeDiskSpace $tempRoot
    $neededBytes = [int64](20 * 1GB)
    $freeGB = [math]::Round($freeBytes / 1GB, 2)
    if ($freeBytes -lt $neededBytes) {
        Write-Host "`n$(T 'NoSpace') (TEMP Drive Free: $freeGB GB / Needed: 20 GB)" -ForegroundColor Red
        Pause-Menu
        return $false
    }

    Write-Host "`n$(T 'DownloadInfo')" -ForegroundColor Yellow
    Write-Host " [1] $(T 'DownloadZip'): $zipPath" -ForegroundColor Gray
    Write-Host " [2] $(T 'DownloadExtract'): $extractRoot" -ForegroundColor Gray
    Write-Host " [3] $(T 'FreeDiskLabel'): $freeGB GB" -ForegroundColor Gray

    Write-Host "`n$(T 'DownloadOptionsPrompt')" -ForegroundColor Yellow
    $ans = $null
    try { $ans = (Read-Host "$(T 'Select') [1/2]").Trim() } catch { }
    if ($ans -ne '1') {
        Write-Host "`n$(T 'ReturningToMenu')" -ForegroundColor Yellow
        Start-Sleep -Milliseconds 600
        return $false
    }

    try {
        # Clean any prior remnants
        Write-Host "`n$(T 'Step1Prep')" -ForegroundColor Cyan
        Remove-DownloadTemp

        # Resolve MediaFire direct link
        Write-Host "`n$(T 'Step2Link') $(T 'Downloading')" -ForegroundColor Cyan
        Write-Host (T 'ConnectingMF') -ForegroundColor DarkGray
        $page = Invoke-WebRequest -Uri $script:MediaFireUrl -UseBasicParsing -ErrorAction Stop
        $urlPattern = 'https://download[^\s"''<>]+'
        $regexMatch = [regex]::Match($page.Content, $urlPattern)
        if (-not $regexMatch.Success) { throw 'MediaFire direct download link not found on page.' }
        $downloadUrl = $regexMatch.Value -replace '&amp;', '&'

        Write-Host (T 'LinkOK') -ForegroundColor DarkCyan
        Write-Host " -> $(T 'DownloadingZip')" -ForegroundColor Cyan

        Invoke-DownloadWithProgressBar -Url $downloadUrl -Destination $zipPath

        if (-not (Test-Path -LiteralPath $zipPath -PathType Leaf)) {
            throw "The ZIP file was not created at: $zipPath"
        }
        $zipSizeGB = [math]::Round(((Get-Item $zipPath).Length / 1GB), 2)
        Write-Host " -> $(T 'ZipDownloaded') ($(T 'ZipSizeLabel'): $zipSizeGB GB)" -ForegroundColor Green

        # Extract archive with live progress
        Write-Host "`n$(T 'Step3Extract') $(T 'Extracting')" -ForegroundColor Cyan
        Write-Host "$(T 'ExtractDest') $extractRoot" -ForegroundColor DarkGray
        if (Test-Path -LiteralPath $extractRoot) {
            Remove-Item $extractRoot -Recurse -Force -ErrorAction SilentlyContinue
        }
        New-Item -ItemType Directory -Force -Path $extractRoot -ErrorAction Stop | Out-Null

        $extractedOk = $false
        try {
            $archive = [System.IO.Compression.ZipFile]::OpenRead($zipPath)
            $totalEntries = $archive.Entries.Count
            $entryIndex = 0
            foreach ($entry in $archive.Entries) {
                $entryIndex++
                $destPath = [System.IO.Path]::Combine($extractRoot, $entry.FullName)
                if ([string]::IsNullOrEmpty($entry.Name)) {
                    if (-not (Test-Path -LiteralPath $destPath)) {
                        [System.IO.Directory]::CreateDirectory($destPath) | Out-Null
                    }
                } else {
                    $parentDir = [System.IO.Path]::GetDirectoryName($destPath)
                    if (-not (Test-Path -LiteralPath $parentDir)) {
                        [System.IO.Directory]::CreateDirectory($parentDir) | Out-Null
                    }
                    [System.IO.Compression.ZipFileExtensions]::ExtractToFile($entry, $destPath, $true)
                }
                if ($totalEntries -gt 0 -and ($entryIndex % 5 -eq 0 -or $entryIndex -eq $totalEntries)) {
                    Show-ModernProgressBar -Current $entryIndex -Total $totalEntries -StatusText $entry.Name -BarColor 'Cyan'
                }
            }
            $archive.Dispose()
            $extractedOk = $true
        }
        catch {
            Expand-Archive -LiteralPath $zipPath -DestinationPath $extractRoot -Force -ErrorAction Stop
            $extractedOk = $true
        }

        if (-not (Test-Path -LiteralPath $extractRoot)) {
            throw "Extraction folder was not created: $extractRoot"
        }
        Write-Host " -> $(T 'ExtractDone')" -ForegroundColor Green

        # Check package structure
        Write-Host "`n$(T 'Step4Check') $(T 'FindingPackage')" -ForegroundColor Cyan
        $candidates = @(Get-Item -LiteralPath $extractRoot -ErrorAction SilentlyContinue) +
        @(Get-ChildItem -LiteralPath $extractRoot -Directory -Recurse -ErrorAction SilentlyContinue)

        $package = $candidates | Where-Object {
            (Test-Path (Join-Path $_.FullName 'storage\t6')) -and
            (Test-Path (Join-Path $_.FullName 'steam\zone')) -and
            (Test-Path (Join-Path $_.FullName 'steam\sound'))
        } | Select-Object -First 1

        if ($null -eq $package) { throw 'Package structure (storage\t6 / steam\zone / steam\sound) not found inside the ZIP.' }
        Write-Host " -> $(T 'PackageFound')" -ForegroundColor Green

        # Move to Desktop
        Write-Host "`n$(T 'Step5Move') $(T 'MovingToDesktop')" -ForegroundColor Cyan
        $desktopPath = [System.Environment]::GetFolderPath('Desktop')
        $destFolder = Join-Path $desktopPath 'ZombiesDeclassified_BETA1'
        Write-Host "$(T 'MoveDest') $destFolder" -ForegroundColor DarkGray
        if (Test-Path $destFolder) {
            Write-Host (T 'ReplacingPrev') -ForegroundColor DarkGray
            Remove-Item $destFolder -Recurse -Force -ErrorAction SilentlyContinue
        }
        Move-Item -LiteralPath $package.FullName -Destination $destFolder -Force -ErrorAction Stop
        Write-Host " -> $(T 'MovedToDesktop') $destFolder" -ForegroundColor Green

        # Clean up temporaries with explicit messages
        Write-Host "`n$(T 'Step6Cleanup')" -ForegroundColor Cyan
        Remove-DownloadTemp
        Write-Host " -> $(T 'TempCleanupDone')" -ForegroundColor Green

        # Set script variables to auto-selected path
        $script:Root = $destFolder
        $script:StorageSource = Join-Path $script:Root 'storage\t6'
        $script:GameSource = Join-Path $script:Root 'steam'

        Write-Host "`n========================================================" -ForegroundColor Green
        Write-Host " $(T 'DownloadDone')" -ForegroundColor Green
        Write-Host "$(T 'FolderLabel') $destFolder" -ForegroundColor Cyan
        Write-Host "========================================================" -ForegroundColor Green

        # Preguntar inmediatamente si desea instalar el mod ahora mismo
        $yesKey = Get-YesKey
        Write-Host "`n========================================================" -ForegroundColor Yellow
        Write-Host " $(T 'WizardActionAskInstall') [$yesKey/N]: " -NoNewline -ForegroundColor Yellow
        $ans = $null
        try { $ans = (Read-Host).Trim().ToUpperInvariant() } catch { }
        if ($ans -eq $yesKey) {
            Invoke-InstallCommon $script:TargetGame
            $script:JustInstalledFromDownload = $true
            Pause-Menu
        } else {
            Write-Host "`n  Puedes instalar el mod en cualquier momento desde el menu." -ForegroundColor Gray
            Start-Sleep -Milliseconds 1200
        }
        return $true

    }
    catch {
        Write-Host (T 'CleanupOnError') -ForegroundColor DarkGray
        Remove-DownloadTemp
        Write-Host "`n$(T 'DownloadError') $($_.Exception.Message)" -ForegroundColor Red
        Pause-Menu
        return $false
    }
}

function Clear-DownloadTemp {
    $tempRoot = Get-TempRoot
    $existingPaths = @(
        (Join-Path $tempRoot 'ZombiesDeclassified_BETA1.zip'),
        (Join-Path $tempRoot 'ZombiesDeclassified_BETA1_package')
    ) | Where-Object { -not [string]::IsNullOrWhiteSpace($_) -and (Test-Path $_) }

    if (-not $existingPaths) {
        Write-Host (T 'CleanupNone') -ForegroundColor Yellow
        return
    }

    Write-Host "`n$(T 'CleanupInfo')" -ForegroundColor Yellow
    $existingPaths | ForEach-Object { Write-Host " -> $_" -ForegroundColor Gray }

    $yesKey = Get-YesKey
    $answer = $null
    try { $answer = (Read-Host "`n$(T 'CleanupAsk') [$yesKey/N]").Trim().ToUpperInvariant() } catch { }
    if ($answer -ne $yesKey) { return }

    Remove-DownloadTemp

    # Fallback to desktop or script root
    $autoPath = Find-PackageAuto
    $fallbackRoot = if ($autoPath) { $autoPath } elseif ($PSScriptRoot -and (Test-Path $PSScriptRoot)) { $PSScriptRoot } else { $script:Root }
    $script:Root = $fallbackRoot
    $script:StorageSource = Join-Path $fallbackRoot 'storage\t6'
    $script:GameSource = Join-Path $fallbackRoot 'steam'

    Write-Host "`n$(T 'CleanupDone')" -ForegroundColor Green
}

# ── Steam Detection ───────────────────────────────────────────────────────────
function Get-SteamGames {
    $roots = [System.Collections.Generic.List[string]]::new()
    $steamInstalled = $false

    $registryPaths = @(
        'HKCU:\Software\Valve\Steam',
        'HKLM:\SOFTWARE\Valve\Steam',
        'HKLM:\SOFTWARE\WOW6432Node\Valve\Steam'
    )
    foreach ($regPath in $registryPaths) {
        try {
            $steamPath = (Get-ItemProperty -Path $regPath -Name SteamPath -ErrorAction Stop).SteamPath
            if (-not [string]::IsNullOrWhiteSpace($steamPath)) {
                $steamInstalled = $true
                $roots.Add($steamPath)
            }
        }
        catch { }
    }

    foreach ($base in @($env:ProgramFiles, ${env:ProgramFiles(x86)})) {
        if (-not [string]::IsNullOrWhiteSpace($base)) { $roots.Add("$base\Steam") }
    }

    $games = [System.Collections.Generic.List[string]]::new()
    foreach ($root in ($roots | Select-Object -Unique)) {
        if ([string]::IsNullOrWhiteSpace($root)) { continue }
        if (Test-Path (Join-Path $root 'steam.exe')) { $steamInstalled = $true }

        $direct = Join-Path $root 'steamapps\common\Call of Duty Black Ops II'
        if (Test-Path $direct) { $games.Add($direct) }

        $vdf = Join-Path $root 'steamapps\libraryfolders.vdf'
        if (Test-Path $vdf) {
            try {
                $vdfContent = Get-Content -LiteralPath $vdf -Raw -ErrorAction SilentlyContinue
                $vdfMatches = [regex]::Matches($vdfContent, '"path"\s+"([^"]+)"')
                foreach ($m in $vdfMatches) {
                    $library = $m.Groups[1].Value -replace '\\\\', '\'
                    $candidate = Join-Path $library 'steamapps\common\Call of Duty Black Ops II'
                    if ((Test-Path $candidate) -and ($games -notcontains $candidate)) {
                        $games.Add($candidate)
                    }
                }
            }
            catch { }
        }
    }
    [PSCustomObject]@{
        Installed = $steamInstalled
        Games     = @($games | Select-Object -Unique)
    }
}

# ── Tree Verification & Installation ──────────────────────────────────────────
function Test-Tree([string]$source, [string]$destination) {
    if (-not (Test-Path -LiteralPath $source)) {
        Write-Host (T 'NoFiles') -ForegroundColor Yellow
        return $false
    }
    $files = @(Get-ChildItem -LiteralPath $source -File -Recurse -ErrorAction SilentlyContinue)
    if ($files.Count -eq 0) { Write-Host (T 'NoFiles') -ForegroundColor Yellow; return $false }

    # Normalize source path so it always ends without trailing slash
    $sourceNorm = $source.TrimEnd('\', '/')

    $different = 0
    $missing = 0
    for ($i = 0; $i -lt $files.Count; $i++) {
        $file = $files[$i]
        # Safe relative path calculation: strip source prefix + exactly one separator
        $fullNorm = $file.FullName
        $relative = $fullNorm.Substring($sourceNorm.Length)
        # Remove exactly one leading backslash (not TrimStart which removes ALL)
        if ($relative.Length -gt 0 -and $relative[0] -eq '\') {
            $relative = $relative.Substring(1)
        }
        $target = Join-Path $destination $relative
        if (-not (Test-Path -LiteralPath $target)) {
            $missing++; $different++
            Write-Host "`n  MISSING / FALTA / AUSENTE: $relative" -ForegroundColor Red
        }
        else {
            $targetFile = Get-Item -LiteralPath $target -Force -ErrorAction SilentlyContinue
            if ($targetFile -and ($targetFile.Length -ne $file.Length)) {
                $different++
                Write-Host "`n  DIFFERENT / DIFERENTE: $relative" -ForegroundColor Yellow
            }
        }
        Show-ModernProgressBar -Current ($i + 1) -Total $files.Count -StatusText $relative -BarColor 'Cyan'
    }
    if ($different -eq 0) {
        Write-Host "   [OK] $(T 'AllGood')" -ForegroundColor Green
        return $true
    }
    Write-Host "$(T 'Problems'): $different. Missing / Faltan / Ausentes: $missing" -ForegroundColor Yellow
    return $false
}

# ── Process Conflict Detection & Robust Copy ─────────────────────────────────
function Get-GameProcesses {
    $targetNames = @(
        't6zm', 't6mp', 't6sp',
        'plutonium-bootstrapper-win32', 'plutonium',
        'BlackOps2', 't6zm.exe', 't6mp.exe', 't6sp.exe'
    )
    $found = [System.Collections.Generic.List[System.Diagnostics.Process]]::new()
    try {
        $procs = Get-Process -ErrorAction SilentlyContinue
        foreach ($p in $procs) {
            foreach ($t in $targetNames) {
                if ($p.ProcessName -like "*$t*") {
                    $found.Add($p)
                    break
                }
            }
        }
    }
    catch { }
    return $found
}

function Resolve-ProcessConflict {
    while ($true) {
        $procs = Get-GameProcesses
        if ($procs.Count -eq 0) {
            return $true
        }

        Write-Host "`n========================================================" -ForegroundColor Red
        Write-Host " [!] $(T 'ProcessConflictTitle')" -ForegroundColor Red
        Write-Host "========================================================" -ForegroundColor Red
        Write-Host (T 'ProcessConflictInfo') -ForegroundColor Yellow
        foreach ($p in $procs) {
            Write-Host "  -> $($p.ProcessName) (PID: $($p.Id))" -ForegroundColor Cyan
        }
        Write-Host "`n$(T 'ProcessConflictAction')" -ForegroundColor Yellow
        $action = $null
        try { $action = (Read-Host (T 'OptionPromptRCM')).Trim().ToUpperInvariant() } catch { }

        switch ($action) {
            'C' {
                Write-Host (T 'ClosingProcs') -ForegroundColor DarkCyan
                foreach ($p in $procs) {
                    try { Stop-Process -Id $p.Id -Force -ErrorAction SilentlyContinue } catch { }
                }
                Start-Sleep -Seconds 1
                Write-Host (T 'ProcessCloseDone') -ForegroundColor Green
            }
            'R' {
                Start-Sleep -Milliseconds 500
            }
            'M' {
                Write-Host "`n$(T 'CopyCancelled')" -ForegroundColor Yellow
                return $false
            }
            default {
                Write-Host (T 'InvalidOptionShort') -ForegroundColor Red
            }
        }
    }
}

# ── Copy a single file from manifest with retry on conflict ──────────────────
function Invoke-CopyFileWithRetry([string]$srcFile, [string]$destFile) {
    while ($true) {
        try {
            $destDir = Split-Path $destFile -Parent
            if (-not (Test-Path $destDir)) {
                New-Item -ItemType Directory -Force -Path $destDir -ErrorAction Stop | Out-Null
            }
            Copy-Item -LiteralPath $srcFile -Destination $destFile -Force -ErrorAction Stop
            return $true
        }
        catch {
            Write-Host "`n[$(T 'CopyErrorLabel')] $($_.Exception.Message)" -ForegroundColor Red
            if (-not (Resolve-ProcessConflict)) { return $false }
            Write-Host "`n$(T 'ProcessCopyFailed')" -ForegroundColor Yellow
            Write-Host "$(T 'ProcessConflictAction')" -ForegroundColor Yellow
            $action = $null
            try { $action = (Read-Host (T 'OptionPromptRM')).Trim().ToUpperInvariant() } catch { }
            if ($action -eq 'M') {
                Write-Host "`n$(T 'CopyCancelled')" -ForegroundColor Yellow
                return $false
            }
        }
    }
}

# ── Copy a subset of the manifest (filtered by prefix) step by step ──────────
function Invoke-CopyManifestGroup {
    param(
        [string]   $StepLabel,       # e.g. '[PASO 1/5] mods\dlc5'
        [string[]] $FileList,        # relative paths to copy
        [string]   $SourceRoot,      # absolute source base
        [string]   $DestRoot,        # absolute destination base
        [string]   $FilterPrefix     # only copy files whose rel path starts with this (empty = all)
    )
    $subset = if ($FilterPrefix) {
        @($FileList | Where-Object { $_.StartsWith($FilterPrefix, [System.StringComparison]::OrdinalIgnoreCase) })
    } else { @($FileList) }

    if ($subset.Count -eq 0) { return $true }

    Write-Host "`n--------------------------------------------------------" -ForegroundColor DarkCyan
    Write-Host " $StepLabel" -ForegroundColor Cyan
    Write-Host "   $(T 'Copying'): $DestRoot\$FilterPrefix" -ForegroundColor DarkGray
    Write-Host "   $($subset.Count) archivos / files" -ForegroundColor DarkGray
    Write-Host "--------------------------------------------------------" -ForegroundColor DarkCyan

    # Check for running game processes ONCE - but don't abort if ignored, just warn
    $procs = Get-GameProcesses
    if ($procs.Count -gt 0) {
        Write-Host "`n  [!] $(T 'ProcessConflictTitle')" -ForegroundColor Yellow
        foreach ($p in $procs) { Write-Host "      $($p.ProcessName) (PID: $($p.Id))" -ForegroundColor DarkYellow }
        Write-Host "  Se intentara copiar de todas formas (los archivos en uso pueden fallar)." -ForegroundColor DarkYellow
    }

    $failed = [System.Collections.Generic.List[string]]::new()

    for ($i = 0; $i -lt $subset.Count; $i++) {
        $rel      = $subset[$i]
        $srcFile  = Join-Path $SourceRoot $rel
        $destFile = Join-Path $DestRoot   $rel

        if (-not (Test-Path -LiteralPath $srcFile -PathType Leaf)) {
            Write-Host "   [SKIP - no existe en fuente] $rel" -ForegroundColor DarkGray
            continue
        }

        Show-ModernProgressBar -Current ($i + 1) -Total $subset.Count -StatusText $rel -BarColor 'Cyan'

        # Create destination directory if needed
        $destDir = Split-Path $destFile -Parent
        if (-not (Test-Path -LiteralPath $destDir)) {
            try {
                New-Item -ItemType Directory -Force -Path $destDir -ErrorAction Stop | Out-Null
            } catch {
                Write-Host "`n   [ERR DIR] No se pudo crear directorio: $destDir" -ForegroundColor Red
                Write-Host "   Error: $($_.Exception.Message)" -ForegroundColor DarkRed
                $failed.Add($rel)
                continue
            }
        }

        # Force-copy the file
        try {
            Copy-Item -LiteralPath $srcFile -Destination $destFile -Force -ErrorAction Stop
        } catch {
            # If it's a file-in-use error, retry up to 3 times then skip
            $ok = $false
            for ($retry = 1; $retry -le 3; $retry++) {
                Start-Sleep -Milliseconds (500 * $retry)
                try {
                    Copy-Item -LiteralPath $srcFile -Destination $destFile -Force -ErrorAction Stop
                    $ok = $true
                    break
                } catch { }
            }
            if (-not $ok) {
                Write-Host "`n   [ERR COPY] $rel" -ForegroundColor Red
                Write-Host "   Error: $($_.Exception.Message)" -ForegroundColor DarkRed
                $failed.Add($rel)
                continue
            }
        }
    }

    if ($failed.Count -gt 0) {
        Write-Host "`n   [!] $($failed.Count) archivo(s) NO copiado(s) en este grupo:" -ForegroundColor Red
        foreach ($f in $failed) { Write-Host "       - $f" -ForegroundColor DarkRed }
        return $false
    }

    Write-Host "   [OK] $(T 'Complete')" -ForegroundColor Green
    return $true
}

function Invoke-VerifyInstallation([string]$game) {
    if (-not (Ensure-SourceReady)) {
        Write-Host (T 'NoSource') -ForegroundColor Red
        return $false
    }
    if ([string]::IsNullOrWhiteSpace($game) -or -not (Test-Path -LiteralPath $game)) {
        if (-not [string]::IsNullOrWhiteSpace($script:TargetGame) -and (Test-Path -LiteralPath $script:TargetGame)) {
            $game = $script:TargetGame
        }
    }
    $found = $false
    $allOk = $true
    if (-not [string]::IsNullOrWhiteSpace($game) -and (Test-Path $game)) {
        $found = $true
        Write-Host "`n$(T 'Verifying') base game: $game" -ForegroundColor Cyan
        $res = Test-Tree $script:GameSource $game
        if (-not $res) { $allOk = $false }
    }
    if (Test-Path $script:PlutoniumStorage) {
        $found = $true
        Write-Host "`n$(T 'Verifying') T6: $script:PlutoniumStorage" -ForegroundColor Cyan
        $res2 = Test-Tree $script:StorageSource $script:PlutoniumStorage
        if (-not $res2) { $allOk = $false }
    }
    if (-not $found) {
        Write-Host (T 'None') -ForegroundColor Yellow
        return $false
    }
    elseif ($allOk) {
        Write-Host "`n$(T 'VerifyComplete')" -ForegroundColor Green
        return $true
    }
    else {
        # Files are missing or different - offer auto-repair right away
        Write-Host "`n========================================================" -ForegroundColor Yellow
        Write-Host " [!] Se detectaron archivos faltantes o con diferencias." -ForegroundColor Yellow
        Write-Host "========================================================" -ForegroundColor Yellow
        $yesKey = Get-YesKey
        Write-Host " Deseas copiar y reparar los archivos faltantes ahora mismo? [$yesKey/N]: " -NoNewline -ForegroundColor Yellow
        $ans = $null
        try { $ans = (Read-Host).Trim().ToUpperInvariant() } catch { }
        if ($ans -eq $yesKey) {
            Write-Host "`n$(T 'Preparing')" -ForegroundColor Cyan
            if (-not (Invoke-InstallStorage)) { return $false }
            if (-not [string]::IsNullOrWhiteSpace($game)) {
                if (-not (Invoke-InstallGame $game)) { return $false }
            }
            Write-Host "`n[OK] $(T 'Complete')" -ForegroundColor Green
            return $true
        }
        return $false
    }
}

function Copy-FolderDirect([string]$SourceDir, [string]$TargetDir, [string]$Label) {
    if (-not (Test-Path -LiteralPath $SourceDir)) {
        Write-Host "   [AVISO] Carpeta de origen no encontrada: $SourceDir" -ForegroundColor Yellow
        return $true
    }
    New-Item -ItemType Directory -Force -Path $TargetDir -ErrorAction SilentlyContinue | Out-Null

    $allFiles = @(Get-ChildItem -LiteralPath $SourceDir -File -Recurse -Force -ErrorAction SilentlyContinue)
    if ($allFiles.Count -eq 0) {
        Write-Host "   [AVISO] No hay archivos en: $SourceDir" -ForegroundColor DarkGray
        return $true
    }

    Write-Host "`n--------------------------------------------------------" -ForegroundColor DarkCyan
    Write-Host " $Label" -ForegroundColor Cyan
    Write-Host "   Destino: $TargetDir" -ForegroundColor DarkGray
    Write-Host "   Total: $($allFiles.Count) archivo(s) a transferir" -ForegroundColor DarkGray
    Write-Host "--------------------------------------------------------" -ForegroundColor DarkCyan

    $srcNorm = $SourceDir.TrimEnd('\', '/')
    for ($i = 0; $i -lt $allFiles.Count; $i++) {
        $f = $allFiles[$i]
        $rel = $f.FullName.Substring($srcNorm.Length).TrimStart('\', '/')
        $dest = Join-Path $TargetDir $rel
        $destParent = Split-Path $dest -Parent
        if (-not (Test-Path -LiteralPath $destParent)) {
            New-Item -ItemType Directory -Force -Path $destParent -ErrorAction SilentlyContinue | Out-Null
        }
        Copy-Item -LiteralPath $f.FullName -Destination $dest -Force -ErrorAction SilentlyContinue

        Show-ModernProgressBar -Current ($i + 1) -Total $allFiles.Count -StatusText $rel -BarColor 'Cyan'
    }
    Write-Host "   [OK] Completado exitosamente." -ForegroundColor Green
    return $true
}

# ── Empty subfolder cleanup (only removes folders that became empty after mod deletion) ──
function Clean-EmptySubDirectories([string]$folder, [string]$rootToPreserve) {
    if (-not (Test-Path -LiteralPath $folder)) { return }
    $dirs = @(Get-ChildItem -LiteralPath $folder -Directory -Recurse -ErrorAction SilentlyContinue)
    $dirs = $dirs | Sort-Object { $_.FullName.Length } -Descending
    foreach ($d in $dirs) {
        # Never remove the root folder itself
        if ($rootToPreserve -and ($d.FullName -ieq $rootToPreserve)) { continue }
        $subItems = @(Get-ChildItem -LiteralPath $d.FullName -Force -ErrorAction SilentlyContinue)
        if ($subItems.Count -eq 0) {
            try { Remove-Item -LiteralPath $d.FullName -Force -ErrorAction SilentlyContinue } catch { }
        }
    }
}

# ── Uninstall: remove only exact mod files from a target folder, step by step ─
function Invoke-UninstallGroup {
    param(
        [string]   $StepLabel,
        [string[]] $FileList,
        [string]   $TargetRoot,
        [string]   $FilterPrefix
    )
    $subset = if ($FilterPrefix) {
        @($FileList | Where-Object { $_.StartsWith($FilterPrefix, [System.StringComparison]::OrdinalIgnoreCase) })
    } else { @($FileList) }

    $toDelete = @($subset | Where-Object { Test-Path -LiteralPath (Join-Path $TargetRoot $_) -PathType Leaf })
    if ($toDelete.Count -eq 0) {
        Write-Host "   [SKIP] No hay archivos del mod en: $FilterPrefix" -ForegroundColor DarkGray
        return 0
    }

    Write-Host "`n  --------------------------------------------------------" -ForegroundColor DarkRed
    Write-Host "  $StepLabel" -ForegroundColor Red
    Write-Host "  $($toDelete.Count) archivo(s) a eliminar / file(s) to delete" -ForegroundColor DarkGray
    Write-Host "  --------------------------------------------------------" -ForegroundColor DarkRed

    $deleted = 0
    for ($i = 0; $i -lt $toDelete.Count; $i++) {
        $rel    = $toDelete[$i]
        $target = Join-Path $TargetRoot $rel
        try {
            Remove-Item -LiteralPath $target -Force -ErrorAction Stop
            $deleted++
        }
        catch {
            Write-Host "`n   [ERR] $rel -> $($_.Exception.Message)" -ForegroundColor Red
        }
        Show-ModernProgressBar -Current ($i + 1) -Total $toDelete.Count -StatusText $rel -BarColor 'Red'
    }
    Write-Host "   [OK] Eliminados / Deleted: $deleted" -ForegroundColor Green
    return $deleted
}

function Invoke-Uninstall {
    Clear-Host
    Show-Banner
    Write-HeaderBox (T 'UninstallTitle') (T 'AuthorText')
    Write-Host ''
    Write-Host (T 'UninstallCheckTitle') -ForegroundColor Cyan
    Write-Host ''

    # ── Detect installed mod files using the exact manifest ───────────────────
    $plutInstalled  = $false
    $gameInstalled  = $false
    $gameFolder     = ''

    # 1. Plutonium storage\t6
    if (Test-Path -LiteralPath $script:PlutoniumStorage) {
        $foundPlut = @($script:DLC5_StorageFiles | Where-Object {
            Test-Path -LiteralPath (Join-Path $script:PlutoniumStorage $_)
        })
        if ($foundPlut.Count -gt 0) {
            $plutInstalled = $true
            Write-Host ((T 'UninstallFoundPlut') + " $script:PlutoniumStorage ($($foundPlut.Count) archivos / files)") -ForegroundColor Green
        }
    }

    # 2. Game folder (TargetGame or Steam)
    $possibleGame = if (-not [string]::IsNullOrWhiteSpace($script:TargetGame)) { $script:TargetGame } else { $script:SteamGame }
    if ([string]::IsNullOrWhiteSpace($possibleGame)) {
        $steamInfo = Get-SteamGames
        $possibleGame = @($steamInfo.Games) | Select-Object -First 1
    }
    if (-not [string]::IsNullOrWhiteSpace($possibleGame) -and (Test-Path -LiteralPath $possibleGame)) {
        $foundGame = @($script:DLC5_GameFiles | Where-Object {
            Test-Path -LiteralPath (Join-Path $possibleGame $_)
        })
        if ($foundGame.Count -gt 0) {
            $gameInstalled = $true
            $gameFolder    = $possibleGame
            Write-Host ((T 'UninstallFoundGame') + " $gameFolder ($($foundGame.Count) archivos / files)") -ForegroundColor Green
        }
    }

    # 3. Manual game folder selection if not auto-found
    if (-not $gameInstalled) {
        $yesKey   = Get-YesKey
        $ansManual = $null
        try { $ansManual = (Read-Host "`n$(T 'UninstallAskManualGame') [$yesKey/N]").Trim().ToUpperInvariant() } catch { }
        if ($ansManual -eq $yesKey) {
            $selected = Select-Folder (T 'Game')
            if (-not [string]::IsNullOrWhiteSpace($selected) -and (Test-Path -LiteralPath $selected)) {
                $foundManual = @($script:DLC5_GameFiles | Where-Object {
                    Test-Path -LiteralPath (Join-Path $selected $_)
                })
                if ($foundManual.Count -gt 0) {
                    $gameInstalled = $true
                    $gameFolder    = $selected
                    Write-Host "`n$((T 'UninstallFoundGame')) $gameFolder ($($foundManual.Count) archivos / files)" -ForegroundColor Green
                } else {
                    Write-Host "`n$(T 'UninstallNoGameFiles')" -ForegroundColor Yellow
                }
            }
        }
    }

    if (-not $plutInstalled -and -not $gameInstalled) {
        Write-Host "`n$(T 'UninstallNone')" -ForegroundColor Yellow
        Pause-Menu
        return
    }

    Write-Host ''
    Write-Host '========================================================' -ForegroundColor Red
    Write-Host ' [!] ' -NoNewline -ForegroundColor Red
    Write-Host (T 'UninstallTitle') -ForegroundColor Red
    Write-Host '========================================================' -ForegroundColor Red
    Write-Host ''

    $didAnything = $false
    $yesKey      = Get-YesKey

    # ── Remove Plutonium storage files step by step ───────────────────────────
    if ($plutInstalled) {
        $ans = $null
        try { $ans = (Read-Host "`n$(T 'UninstallAskPlut') [$yesKey/N]").Trim().ToUpperInvariant() } catch { }
        if ($ans -eq $yesKey) {
            Write-Host "`n$(T 'UninstallDeleting') $script:PlutoniumStorage" -ForegroundColor DarkRed
            Write-Host '  (Solo archivos del mod / Mod files only - otros archivos NO se tocan)' -ForegroundColor DarkGray
            $totalPlut = 0

            $storageGroups = @(
                @{ Label = '[1/5] mods\dlc5 (archivos del mod)';       Prefix = 'mods\dlc5' }
                @{ Label = '[2/5] raw\clientscripts';                   Prefix = 'raw\clientscripts' }
                @{ Label = '[3/5] raw\maps\mp\animscripts';             Prefix = 'raw\maps' }
                @{ Label = '[4/5] raw\scripts\zm (scripts)';           Prefix = 'raw\scripts' }
                @{ Label = '[5/5] zone (archivos .ff de mapas)';        Prefix = 'zone' }
            )
            foreach ($g in $storageGroups) {
                $totalPlut += Invoke-UninstallGroup `
                    -StepLabel    $g.Label `
                    -FileList     $script:DLC5_StorageFiles `
                    -TargetRoot   $script:PlutoniumStorage `
                    -FilterPrefix $g.Prefix
            }

            # Remove empty mod-created subfolders only (never t6 root itself)
            Clean-EmptySubDirectories (Join-Path $script:PlutoniumStorage 'mods') $script:PlutoniumStorage
            Clean-EmptySubDirectories (Join-Path $script:PlutoniumStorage 'raw')  $script:PlutoniumStorage
            Clean-EmptySubDirectories (Join-Path $script:PlutoniumStorage 'zone') $script:PlutoniumStorage

            Write-Host "`n  Total eliminados Plutonium: $totalPlut (carpeta t6 preservada intacta)." -ForegroundColor Green
            $didAnything = $true
        }
    }

    # ── Remove game files step by step ───────────────────────────────────────
    if ($gameInstalled) {
        $ans = $null
        try { $ans = (Read-Host "`n$(T 'UninstallAskGame') [$yesKey/N]").Trim().ToUpperInvariant() } catch { }
        if ($ans -eq $yesKey) {
            Write-Host "`n$(T 'UninstallDeleting') $gameFolder" -ForegroundColor DarkRed
            Write-Host '  (Solo archivos del mod / Mod files only - otros archivos NO se tocan)' -ForegroundColor DarkGray
            $totalGame = 0

            $gameGroups = @(
                @{ Label = '[1/2] sound (archivos de audio)';   Prefix = 'sound' }
                @{ Label = '[2/2] zone\all (archivos .ipak)';   Prefix = 'zone\all' }
            )
            foreach ($g in $gameGroups) {
                $totalGame += Invoke-UninstallGroup `
                    -StepLabel    $g.Label `
                    -FileList     $script:DLC5_GameFiles `
                    -TargetRoot   $gameFolder `
                    -FilterPrefix $g.Prefix
            }

            # Remove empty mod-created subfolders only
            Clean-EmptySubDirectories (Join-Path $gameFolder 'sound') $gameFolder
            Clean-EmptySubDirectories (Join-Path $gameFolder 'zone')  $gameFolder

            Write-Host "`n  Total eliminados juego: $totalGame." -ForegroundColor Green
            $didAnything = $true
        }
    }

    Write-Host ''
    if ($didAnything) {
        Write-Host (T 'UninstallComplete') -ForegroundColor Green
    } else {
        Write-Host (T 'UninstallCancelled') -ForegroundColor Yellow
    }
    Pause-Menu
}

function Write-HeaderBox {
    param([string]$line1Text, [string]$line2Text)
    $inner = 64
    $border = '+' + ('-' * $inner) + '+'

    $p1Left = [math]::Max(0, [math]::Floor(($inner - $line1Text.Length) / 2))
    $p1Right = [math]::Max(0, $inner - $line1Text.Length - $p1Left)
    $row1 = '|' + (' ' * $p1Left) + $line1Text + (' ' * $p1Right) + '|'

    $p2Left = [math]::Max(0, [math]::Floor(($inner - $line2Text.Length) / 2))
    $p2Right = [math]::Max(0, $inner - $line2Text.Length - $p2Left)
    $row2 = '|' + (' ' * $p2Left) + $line2Text + (' ' * $p2Right) + '|'

    Write-Host $border -ForegroundColor DarkCyan
    Write-Host $row1   -ForegroundColor Cyan
    Write-Host $row2   -ForegroundColor DarkCyan
    Write-Host $border -ForegroundColor DarkCyan
}

function Step-SelectLanguage {

# ── WIZARD STEP 1: Game Version Selection (Steam or Alternative) ──────────────
function Step-SelectGameVersion {
    while ($true) {
        Clear-Host
        Show-Banner
        Write-HeaderBox (T 'WizardWelcome') (T 'AuthorText')

        Write-Host "`n========================================================" -ForegroundColor Cyan
        Write-Host " $(T 'WizardStepGame')" -ForegroundColor Cyan
        Write-Host "========================================================" -ForegroundColor Cyan
        Write-Host "`n $(T 'WizardGameModePrompt')" -ForegroundColor Yellow
        Write-Host "`n  [1] $(T 'WizardOptionSteam')" -ForegroundColor White
        Write-Host "  [2] $(T 'WizardOptionAlt')" -ForegroundColor White
        Write-Host "  [9] $(T 'About')" -ForegroundColor Magenta
        Write-Host "  [3] $(T 'Exit')" -ForegroundColor DarkGray

        $mode = $null
        while ($mode -notmatch '^[1-39]$') {
            try { $mode = (Read-Host "`n$(T 'Select') [1/2/3/9]").Trim() } catch { }
        }

        if ($mode -eq '9') {
            Show-About
            continue
        }

        if ($mode -eq '3') { return $false }

        $script:TargetGame = $null
        if ($mode -eq '1') {
            # Steam original mode
            $steamInfo = Get-SteamGames
            $gamePath = @($steamInfo.Games) | Select-Object -First 1
            if (-not [string]::IsNullOrWhiteSpace($gamePath) -and (Test-Path -LiteralPath $gamePath)) {
                $script:TargetGame = $gamePath
                Write-Host "`n$(T 'WizardSteamAutoFound') $gamePath" -ForegroundColor Green
                Start-Sleep -Milliseconds 900
                return $true
            } else {
                Write-Host "`n$(T 'WizardSteamNotFound')" -ForegroundColor Yellow
                $yesKey = Get-YesKey
                $ask = $null
                try { $ask = (Read-Host "`n$(T 'WizardSteamAskManual') [$yesKey/N]").Trim().ToUpperInvariant() } catch { }
                if ($ask -eq $yesKey) {
                    $selected = Select-Folder (T 'BaseGame')
                    if (-not [string]::IsNullOrWhiteSpace($selected) -and (Test-Path -LiteralPath $selected)) {
                        $script:TargetGame = $selected
                        Write-Host "`nBlack Ops II: $selected" -ForegroundColor Green
                        Start-Sleep -Milliseconds 800
                        return $true
                    }
                }
                Write-Host "`n$(T 'CancelFolderReturn')" -ForegroundColor Yellow
                Start-Sleep -Milliseconds 800
                return $false
            }
        } else {
            # Alternative mode
            Write-Host "`n$(T 'WizardAltFolderPrompt')" -ForegroundColor Cyan
            Write-Host "$(T 'GameHelp')" -ForegroundColor Gray
            $selected = Select-Folder (T 'Game')
            if (-not [string]::IsNullOrWhiteSpace($selected) -and (Test-Path -LiteralPath $selected)) {
                $script:TargetGame = $selected
                Write-Host "`nJuego / Game: $selected" -ForegroundColor Green
                Start-Sleep -Milliseconds 800
                return $true
            } else {
                Write-Host "`n$(T 'CancelFolderReturn')" -ForegroundColor Yellow
                Start-Sleep -Milliseconds 800
                return $false
            }
        }
    }
}

# ── WIZARD STEP 2 & 3: Check System Status & Ensure Mod Package ───────────────
function Step-EnsurePackageAndCheckStatus {
    while ($true) {
        Clear-Host
        Show-Banner
        Write-HeaderBox (T 'WizardWelcome') (T 'AuthorText')

        Write-Host "`n========================================================" -ForegroundColor Cyan
        Write-Host " $(T 'WizardStepStatus')" -ForegroundColor Cyan
        Write-Host "========================================================" -ForegroundColor Cyan

        $plutInstalled = Test-DLC5InstalledInStorage
        $gameInstalled = Test-DLC5InstalledInGame $script:TargetGame

        $plutText = if ($plutInstalled) { (T 'WizardStatusInstalled') } else { (T 'WizardStatusNotInst') }
        $plutColor = if ($plutInstalled) { 'Green' } else { 'Yellow' }
        Write-Host "`n  * $(T 'WizardStatusPlutMod') " -NoNewline -ForegroundColor Gray
        Write-Host "[$plutText]" -ForegroundColor $plutColor

        $gameText = if ($gameInstalled) { (T 'WizardStatusInstalled') } else { (T 'WizardStatusNotInst') }
        $gameColor = if ($gameInstalled) { 'Green' } else { 'Yellow' }
        Write-Host "  * $(T 'WizardStatusGameMod') " -NoNewline -ForegroundColor Gray
        Write-Host "[$gameText] ($($script:TargetGame))" -ForegroundColor $gameColor

        # Auto detect package
        $pkgPath = Find-PackageAuto
        if ($pkgPath) {
            Write-Host "  * $(T 'WizardStatusPkg') " -NoNewline -ForegroundColor Gray
            Write-Host "[OK] $pkgPath" -ForegroundColor Green
            $script:Root = $pkgPath
            $script:StorageSource = Join-Path $pkgPath 'storage\t6'
            $script:GameSource = Join-Path $pkgPath 'steam'
            return $true
        } else {
            Write-Host "  * $(T 'WizardStatusPkg') " -NoNewline -ForegroundColor Gray
            Write-Host "[$(T 'WizardPkgNotDetected')]" -ForegroundColor Yellow
        }

        # If package not found, ask user to download or select
        Write-Host "`n========================================================" -ForegroundColor Yellow
        Write-Host " $(T 'WizardStepPackage')" -ForegroundColor Yellow
        Write-Host "========================================================" -ForegroundColor Yellow
        Write-Host "`n $(T 'WizardPkgPrompt')" -ForegroundColor Yellow
        Write-Host "`n  [1] $(T 'WizardPkgOptDownload')" -ForegroundColor White
        Write-Host "  [2] $(T 'WizardPkgOptSelect')" -ForegroundColor White
        Write-Host "  [9] $(T 'About')" -ForegroundColor Magenta
        Write-Host "  [3] $(T 'Exit')" -ForegroundColor DarkGray

        $pkgChoice = $null
        while ($pkgChoice -notmatch '^[1-39]$') {
            try { $pkgChoice = (Read-Host "`n$(T 'Select') [1/2/3/9]").Trim() } catch { }
        }

        if ($pkgChoice -eq '9') {
            Show-About
            continue
        }

        switch ($pkgChoice) {
            '1' {
                Clear-Host
                $ok = Invoke-DownloadPackage
                if (-not $ok) { return $false }
                return $true
            }
            '2' {
                Write-Host "`n$(T 'Package')" -ForegroundColor Cyan
                Write-Host "$(T 'PackageHelp')" -ForegroundColor Gray
                $selected = Select-Folder (T 'Package')
                if ([string]::IsNullOrWhiteSpace($selected) -or -not (Test-PackageFolder $selected)) {
                    Write-Host "`n$(T 'InvalidPackage')" -ForegroundColor Red
                    Start-Sleep -Milliseconds 1200
                    return $false
                }
                $script:Root = $selected
                $script:StorageSource = Join-Path $selected 'storage\t6'
                $script:GameSource = Join-Path $selected 'steam'
                Write-Host "`n$(T 'PackageFound') $selected" -ForegroundColor Green
                Start-Sleep -Milliseconds 800
                return $true
            }
            '3' {
                return $false
            }
        }
    }
}

# ── WIZARD STEP 4: Install, Repair, or Uninstall ──────────────────────────────
function Step-InstallOrManage {
    $script:JustInstalledFromDownload = $false

    while ($true) {
        Clear-Host
        Show-Banner
        Write-HeaderBox (T 'WizardWelcome') (T 'AuthorText')

        Write-Host "`n========================================================" -ForegroundColor Cyan
        Write-Host " $(T 'WizardStepAction')" -ForegroundColor Cyan
        Write-Host "========================================================" -ForegroundColor Cyan

        # Auto-detect game if not set
        if ([string]::IsNullOrWhiteSpace($script:TargetGame) -or -not (Test-Path -LiteralPath $script:TargetGame)) {
            $steamInfo = Get-SteamGames
            foreach ($g in $steamInfo.Games) {
                if (Test-Path -LiteralPath $g) {
                    if (Test-DLC5InstalledInGame $g) {
                        $script:TargetGame = $g
                        break
                    }
                    if ([string]::IsNullOrWhiteSpace($script:TargetGame)) {
                        $script:TargetGame = $g
                    }
                }
            }
        }

        # Auto-detect package if not set
        if (-not (Test-SourceReady)) {
            $pkg = Find-PackageAuto
            if ($pkg) {
                $script:Root = $pkg
                $script:StorageSource = Join-Path $pkg 'storage\t6'
                $script:GameSource = Join-Path $pkg 'steam'
            }
        }

        $plutInstalled = Test-DLC5InstalledInStorage
        $gameInstalled = if (-not [string]::IsNullOrWhiteSpace($script:TargetGame)) { Test-DLC5InstalledInGame $script:TargetGame } else { $false }
        $isInstalled = ($plutInstalled -or $gameInstalled)

        if ($isInstalled) {
            Write-Host "`n $(T 'WizardActionAlready')" -ForegroundColor Yellow
            Write-Host "`n  [1] $(T 'WizardActionRepair')" -ForegroundColor White
            Write-Host "  [2] $(T 'WizardActionUninstall')" -ForegroundColor DarkRed
            Write-Host "  [3] $(T 'WizardActionVerify')" -ForegroundColor White
            Write-Host "  [9] $(T 'About')" -ForegroundColor Magenta
            Write-Host "  [4] $(T 'WizardActionExit')" -ForegroundColor DarkGray

            $act = $null
            while ($act -notmatch '^[1-49]$') {
                try { $act = (Read-Host "`n$(T 'Select') [1-4/9]").Trim() } catch { }
            }

            if ($act -eq '9') {
                Show-About
                continue
            }

            switch ($act) {
                '1' {
                    if (-not (Ensure-SourceReady)) {
                        Write-Host "`n$(T 'NoSource')" -ForegroundColor Red
                        Pause-Menu
                        continue
                    }
                    if ([string]::IsNullOrWhiteSpace($script:TargetGame) -or -not (Test-Path -LiteralPath $script:TargetGame)) {
                        $gOk = Step-SelectGameVersion
                        if (-not $gOk) { continue }
                    }
                    Invoke-InstallCommon $script:TargetGame
                    Pause-Menu
                    continue
                }
                '2' {
                    Invoke-Uninstall
                    continue
                }
                '3' {
                    if (-not (Ensure-SourceReady)) {
                        Write-Host "`n$(T 'NoSource')" -ForegroundColor Red
                        Pause-Menu
                        continue
                    }
                    Invoke-VerifyInstallation $script:TargetGame
                    Pause-Menu
                    continue
                }
                '4' {
                    return
                }
            }
        } else {
            # Mod not installed yet
            $yesKey = Get-YesKey
            Write-Host "`n $(T 'WizardActionAskInstall') [$yesKey/N] (o [9] $(T 'About')): " -NoNewline -ForegroundColor Yellow
            $ans = $null
            try { $ans = (Read-Host).Trim().ToUpperInvariant() } catch { }
            if ($ans -eq '9') {
                Show-About
                continue
            }
            if ($ans -eq $yesKey) {
                if (-not (Ensure-SourceReady)) {
                    Write-Host "`n$(T 'NoSource')" -ForegroundColor Red
                    Pause-Menu
                    continue
                }
                if ([string]::IsNullOrWhiteSpace($script:TargetGame) -or -not (Test-Path -LiteralPath $script:TargetGame)) {
                    $gOk = Step-SelectGameVersion
                    if (-not $gOk) { continue }
                }
                Invoke-InstallCommon $script:TargetGame
                Pause-Menu
                continue
            } else {
                Write-Host "`n$(T 'UninstallCancelled')" -ForegroundColor Yellow
                Start-Sleep -Milliseconds 800
                return
            }
        }
    }
}

# ── Wizard Controller ─────────────────────────────────────────────────────────
function Start-Wizard {
    # Paso 0: Idioma al iniciar
    Step-SelectLanguage

    # Auto-detectar juego (Steam o rutas comunes)
    if ([string]::IsNullOrWhiteSpace($script:TargetGame)) {
        $steamInfo = Get-SteamGames
        foreach ($g in $steamInfo.Games) {
            if (Test-Path -LiteralPath $g) {
                if (Test-DLC5InstalledInGame $g) {
                    $script:TargetGame = $g
                    break
                }
                if ([string]::IsNullOrWhiteSpace($script:TargetGame)) {
                    $script:TargetGame = $g
                }
            }
        }
    }

    # Auto-detectar paquete en Escritorio o directorio del script
    $pkg = Find-PackageAuto
    if ($pkg) {
        $script:Root = $pkg
        $script:StorageSource = Join-Path $pkg 'storage\t6'
        $script:GameSource = Join-Path $pkg 'steam'
    }

    # Comprobar si el mod YA está instalado
    $plutInstalled = Test-DLC5InstalledInStorage
    $gameInstalled = if (-not [string]::IsNullOrWhiteSpace($script:TargetGame)) { Test-DLC5InstalledInGame $script:TargetGame } else { $false }

    if ($plutInstalled -or $gameInstalled) {
        # Si ya está instalado, entrar DIRECTAMENTE al menú de gestión (verificar, desinstalar, reparar)
        Step-InstallOrManage
        return
    }

    # Si NO está instalado, ejecutar el asistente de instalación paso a paso
    while ($true) {
        # Paso 1: Version de juego (Steam o Alternativo)
        $step1Ok = Step-SelectGameVersion
        if (-not $step1Ok) {
            if (Confirm-Exit) { break } else { continue }
        }

        # Paso 2 y 3: Deteccion de estado del sistema y asegurar paquete
        $step2Ok = Step-EnsurePackageAndCheckStatus
        if (-not $step2Ok) {
            if (Confirm-Exit) { break } else { continue }
        }

        # Paso 4: Instalar y quedar dentro del menú de gestión
        Step-InstallOrManage
        break
    }
}
